# 環境変数テンプレート
# 
# このファイルをコピーして、プロジェクトルートに .env ファイルを作成してください。
# 秘密情報を含む値は、実際の値に書き換えてください。
#
# 使用方法:
# 1. このファイルをコピーしてプロジェクトルートに .env ファイルとして配置
#    cp env.example.txt .env
# 2. .env ファイルを編集して、各環境変数の値を実際の値に書き換える
# 3. アプリケーション起動時に自動的に .env ファイルが読み込まれます
#
# 注意: .env ファイルは Git にコミットされません（.gitignore に含まれています）

# ============================================
# データベース接続設定
# ============================================
# PostgreSQL データベースの接続情報
# 秘密情報: 本番環境では必ず書き換えてください

# Docker Compose用の設定
# PostgreSQLコンテナのデータベース名
POSTGRES_DB=super_shiharai_kun

# PostgreSQLコンテナのポート
POSTGRES_PORT=5432

# アプリケーション接続用の設定
# アプリケーションからPostgreSQLに接続する際のホスト名
DB_HOST=localhost

# アプリケーションからPostgreSQLに接続する際のポート
DB_PORT=5432

# アプリケーションからPostgreSQLに接続する際のデータベース名
DB_NAME=super_shiharai_kun

# アプリケーションからPostgreSQLに接続する際のユーザー名
# 注意: docker-compose.ymlでは固定でpostgresユーザーを使用
DB_USERNAME=postgres

# アプリケーションからPostgreSQLに接続する際のパスワード
# 注意: docker-compose.ymlでは固定でpostgresパスワードを使用
DB_PASSWORD=postgres

# ============================================
# JWT 設定
# ============================================
# JWT トークンの署名に使用する秘密鍵
# 秘密情報: 本番環境では必ず強力なランダム文字列に書き換えてください
# 生成方法の例: openssl rand -base64 64
JWT_SECRET=your-secret-key-here-change-this-in-production

# JWT トークンの有効期限（秒）
# デフォルト: 86400 (24時間)
JWT_EXPIRATION=86400

# ============================================
# アプリケーション設定
# ============================================
# サーバーポート
SERVER_PORT=8080

# ログレベル（DEBUG, INFO, WARN, ERROR）
LOGGING_LEVEL=INFO
